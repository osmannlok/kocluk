# 05 Çekirdek Sözleşme (TASLAK, bağlayıcı)
Config: static load(string $basePath): void; get(string $key, mixed $default=null) ('app.name'); basePath(string $sub=''): string. Fonksiyon env(string $k, mixed $d=null): .env değeri (true/false/null/sayı dönüşümlü; kendi ayrıştırıcın).
Database (PDO singleton): pdo(): PDO; fetchAll(string $sql, array $p=[]): array; fetchOne(...): ?array; fetchValue(...): mixed; execute(...): int; insert(...): int; transaction(callable $fn): mixed (iç içe = mevcut transaction'a katılır; hata = rollback + rethrow).
Request: static capture(): self; method(): string (POST'ta _method ile PUT/DELETE); path(): string; input(string $k, $d=null); all(): array; query(?string $k=null, $d=null); post(...); json(): array; file(string $k): ?array; ip(): string (TRUSTED_PROXIES'e göre); userAgent(): string; isAjax(): bool; wantsJson(): bool; header(string $n): ?string; routeParam(string $k, $d=null); setRouteParams(array $p): void.
Response: static html(string $b, int $s=200); json(array $d, int $s=200); redirect(string $url, int $s=302) (açık yönlendirme korumalı); text(string $b, int $s=200); withHeader(string $k, string $v): self; send(): void.
Session: static start(): void; get(string $k, $d=null); put(string $k, $v); has; forget; flash(string $k, $v); pullFlash(string $k, $d=null); regenerate(): void; destroy(): void; id(): string.
Csrf: static token(): string; validate(?string $t): bool; field(): string (name="_token"). AJAX başlığı X-CSRF-Token.
Validator: static make(array $data, array $rules, array $messages=[]): self; fails(): bool; errors(): array; validated(): array. Kurallar: required, nullable, string, integer, numeric, boolean, array, email, min:n, max:n, between:a,b, in:a,b, regex:/../, confirmed, phone_tr, tc_identity, date, url, slug, unique:tablo,kolon[,exceptId], exists:tablo,kolon.
View: static render(string $name, array $data=[], ?string $layout=null): string (layout $content alır); partial(string $name, array $data=[]): string; share(string $k, mixed $v): void.
Router: get/post/put/delete(string $path, array|callable $handler, array $opts=[]) (opts: middleware[]); group(array $attrs[prefix, middleware], callable $fn); {id}, {slug}, {id:\d+}; dispatch(Request $r): Response. Handler [Controller::class,'metot']: Request + adıyla eşleşen route parametreleri. 404/405 HttpException.
Middleware: interface App\Middleware\MiddlewareInterface { handle(Request $r, callable $next, array $args=[]): Response }. Takma adlar (config/middleware.php): auth, guest, perm (perm:product.create; 'a,b' = herhangi biri), throttle (throttle:5,60), csrf, secure_headers. GLOBAL: secure_headers, csrf.
Auth: static check(): bool; user(): ?array (id, role_id, role_name, first_name, last_name, email, status, email_verified_at); id(): ?int; can(string $perm): bool; hasRole(string $r): bool; login(int $userId): void; logout(): void. Her istekte DB'den tazelenir; status != active ise oturum düşer.
RateLimiter: static tooManyAttempts(string $key, int $max, int $seconds): bool; hit(string $key, int $seconds): void; clear(string $key): void; availableIn(string $key): int.
Logger: static error/warning/info(string $msg, array $ctx=[]): void (hassas anahtarlar maskeli).
HttpException(int $status, string $message=''): getStatus(). ValidationException(array $errors, array $old=[]): errors(), old().
ErrorHandler: static register(): void; static handle(Throwable $e, Request $r): Response. HttpException => durum + view('errors.error', [code,title,message]) (view yoksa düz metin); ValidationException => JSON ise 422 {"ok":false,"error":{"code":"validation","errors":{...}}}, değilse flash errors+old ve geri yönlendirme; diğerleri => Logger + 500 (APP_DEBUG=false ise genel mesaj).
Money (App\Helpers\Money): static toCents(string|int|float $amount): int; fromCents(int $cents): string ("1234.50"); percent(int $cents, string $rate): int (yarım yukarı); format(string $decimal): string ("1.234,50 ₺").
functions.php: env, config, e, url, asset, route, redirect (Response döner), back, old(k,d), errors(?k), flash(string $type, string $msg), flash_messages(): array, csrf_field, csrf_token, method_field, auth, can, money, slugify (Türkçe), now, ip_to_bin, bin_to_ip, abort(int $code, string $msg=''), is_path(string $pattern).
config/security.php: ['csp'=>[direktif=>[kaynaklar]], 'csrf_except'=>['/odeme/paytr/bildirim'], 'hsts'=>bool]. config/app.php: name, env, debug, url, key, timezone, trusted_proxies, session[cookie_name, idle_minutes, absolute_minutes], upload[max_mb], mail[...].
BaseController: view(string $name, array $d=[], string $layout='layouts.main'): Response; json(array,int); ok(array $data=[]): Response; fail(string $code, string $msg, int $s=400, array $errors=[]): Response; redirect(string $url): Response; validate(Request $r, array $rules, array $messages=[]): array (başarısızsa ValidationException); flash(string $type, string $msg).
